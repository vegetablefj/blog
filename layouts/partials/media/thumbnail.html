{{- $page := .Page -}} {{- $src := .Src -}} {{- $image := $page.Resources.GetMatch $src -}} {{- if not $image -}} {{- errorf "thumbnail: %q not found in page bundle %q" $src $page.File.Path -}} {{- else -}} {{- $thumbWidth := int (.ThumbWidth | default 960) -}} {{- $thumb := $image -}} {{- if gt $image.Width $thumbWidth -}} {{- $thumb = $image.Resize (printf "%dx webp q80" $thumbWidth) -}} {{- end -}} {{- $alt := .Alt | default $src -}} {{ with .Caption }}{{ end }} {{- end -}}
